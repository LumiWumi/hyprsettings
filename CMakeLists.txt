cmake_minimum_required(VERSION 3.16)

project(hyprsettings VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Qt6 REQUIRED COMPONENTS Widgets)

qt_standard_project_setup()

qt_add_executable(hyprsettings
  src/main.cpp
  src/window.cpp
  src/sidebar.cpp
  include/window.h
  include/sidebar.h
)

target_link_libraries(hyprsettings PRIVATE Qt6::Widgets)

target_include_directories(hyprsettings PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
